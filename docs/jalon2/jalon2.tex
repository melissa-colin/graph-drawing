%---------------------------------------------------------------------
\section{Contexte}

%---------------------------------------------------------------------
\begin{frame}{Le sujet}
\centering
\textbf{Optimiser la position des nœuds d'un graphe directement en espace écran,}\\\textbf{avec des coûts calculés sur le dessin rendu en pixels.}

\vspace{0.6cm}
\begin{tikzpicture}[node distance=0.5cm]
  \node[carte,text width=4.0cm] (e) {\textcolor{bleu}{\scriptsize ENTRÉE}\\\textbf{Un graphe et une taille d'image}\\[2pt]\scriptsize arêtes ; en option taille des nœuds, largeur des arêtes et couleurs};
  \node[carte,text width=4.0cm,right=of e] (s) {\textcolor{bleu}{\scriptsize SORTIE}\\\textbf{Des positions}\\[2pt]\scriptsize $(x, y)$ pour chaque nœud, et éventuellement l'image};
  \node[carte,text width=4.0cm,right=of s] (c) {\textcolor{orange}{\scriptsize MOYEN}\\\textbf{Un rendu différentiable}\\[2pt]\scriptsize nœuds et arêtes dessinés avec des SDF, d'où des coûts écran dérivables};
\end{tikzpicture}

\vspace{0.5cm}
\small Question centrale du sujet : des coûts définis à l'écran donnent-ils des dessins plus fidèles au rendu final que des coûts géométriques ?
\end{frame}

%---------------------------------------------------------------------
\begin{frame}{Un rapide état de l'art}
\small
\begin{tabularx}{\textwidth}{@{}l X@{}}
\toprule
\textbf{Méthode} & \textbf{Principe} \\
\midrule
Kamada-Kawai, stress & optimisation classique, refaite pour chaque graphe \\
(GD)$^2$ & descente de gradient sur des critères esthétiques, graphe par graphe \\
DeepGD & un GNN appris une fois, entraîné sur le stress \\
SmartGD & un GAN qui imite des dessins de référence \\
CoRe-GD (ICLR 2024) & un GNN récurrent qui se recâble selon son propre dessin ; meilleur stress sur Rome \\
\bottomrule
\end{tabularx}

\vspace{0.5cm}
\textcolor{gris}{Toutes optimisent dans un espace géométrique, où un nœud est un point et une arête un segment sans épaisseur. Les pixels n'arrivent qu'à l'affichage, et cette discrétisation peut dégrader la qualité du dessin.}
\end{frame}

\begin{frame}{L'architecture : un GNN récurrent et une sortie en pixels}
\centering
\resizebox{\textwidth}{!}{%
\begin{tikzpicture}[node distance=0.3cm]
  \node[carte] (enc) {\textbf{Encodeur}\\\scriptsize un MLP par nœud};
  \node[carte,right=0.55cm of enc] (mp) {\textbf{Messages}\\\scriptsize le long des arêtes};
  \node[carte,right=of mp] (br) {\textbf{Dessin provisoire}\\\scriptsize par la soft-argmax};
  \node[carte,right=of br] (knn) {\textbf{Messages}\\\scriptsize entre voisins\\[-2pt]\scriptsize dans le dessin};
  \node[carte,right=0.55cm of knn,fill=bleuclair,draw=bleu] (sa) {\textbf{Soft-argmax}\\\scriptsize positions en pixels};
  \node[carte,right=0.55cm of sa,fill=orangeclair,draw=orange,dashed] (af) {\textbf{Affinage}\\\scriptsize optionnel :\\[-2pt]\scriptsize 25 pas d'Adam\\[-2pt]\scriptsize sur les positions};
  \node[draw=bleu,dashed,rounded corners=6pt,fit=(mp)(br)(knn),inner sep=7pt,label={[font=\scriptsize,bleu]above:répété environ 5 fois}] (boucle) {};
  \draw[fleche] (enc) -- (boucle);
  \draw[fleche] (mp) -- (br);
  \draw[fleche] (br) -- (knn);
  \draw[fleche] (boucle) -- (sa);
  \draw[fleche,dashed,orange] (sa) -- (af);
\end{tikzpicture}}

\vspace{0.4cm}
\small
\begin{itemize}
\setlength{\itemsep}{2pt}
\item \textbf{Dessin provisoire} : à chaque tour, la tête soft-argmax place les nœuds à partir de leur état actuel. Ce dessin sert seulement à trouver les 8 nœuds les plus proches de chacun.
\item \textbf{Messages entre voisins dans le dessin} (repris de CoRe-GD) : deux nœuds proches à l'écran mais loin dans le graphe se voient, et peuvent apprendre à s'écarter.
\item \textbf{Lien avec la loss} : les métriques sont calculées sur les positions finales seulement ; leur gradient remonte par la soft-argmax et les messages jusqu'aux poids.
\item \textbf{Affinage (optionnel)} : après le modèle, quelques pas de descente de gradient sur les positions elles-mêmes, pour retirer les petits défauts (voir les résultats).
\end{itemize}
\end{frame}

\begin{frame}{La loss finale : trois étapes, trois dosages}
\centering
\resizebox{\textwidth}{!}{%
\begin{tikzpicture}[node distance=0.5cm,etape/.style={carte,text width=4.2cm,minimum height=3.3cm,align=left}]
  \node[etape,fill=bleuclair,draw=bleu] (p1) {\textbf{1. Entraînement}\\[2pt]
    \scriptsize 30 epochs, lr $2\cdot10^{-4}$\\[4pt]
    \footnotesize\textbf{Stress en pixels seul}\\[4pt]
    \scriptsize\textcolor{gris}{Pourquoi : le stress donne la structure globale. Tout mélanger dès le départ échoue (stress $\times 3$, croisements $\times 2{,}4$).}};
  \node[etape,right=of p1] (p2) {\textbf{2. Fine-tuning}\\[2pt]
    \scriptsize 8 epochs, lr $10^{-4}$\\[4pt]
    \footnotesize\textbf{Stress en pixels + stress à l'échelle optimale + métriques écran} (3\,\% chacune)\\[4pt]
    \scriptsize\textcolor{gris}{Pourquoi : corriger les défauts locaux sans casser la structure.}};
  \node[etape,right=of p2,fill=orangeclair,draw=orange,dashed] (p3) {\textbf{3. Affinage} \scriptsize(optionnel)\\[2pt]
    \scriptsize par dessin, 25 pas d'Adam\\[4pt]
    \footnotesize\textbf{Stress en pixels + métriques écran}\\[4pt]
    \scriptsize\textcolor{gris}{Pourquoi : retirer les derniers défauts de la taille d'un nœud, sans réentraîner.}};
  \draw[fleche] (p1) -- (p2); \draw[fleche] (p2) -- (p3);
\end{tikzpicture}}

\vspace{0.3cm}
\begin{minipage}{0.94\textwidth}
\footnotesize
\begin{itemize}
\setlength{\itemsep}{2pt}
\item \textbf{Deux stress au fine-tuning} : avec le stress à l'échelle optimale seul, les dessins rétrécissaient de 24\,\% (rien ne fixe la taille, et un petit dessin a des défauts plus petits en pixels). Le stress en pixels fixe la taille.
\item \textbf{Poids choisis sur la validation} : 3\,\% est le plus fort dosage qui coûte moins de 5\,\% de stress.
\end{itemize}
\end{minipage}
\end{frame}

\vspace{0.15cm}
\begin{minipage}{0.94\textwidth}
\footnotesize
\begin{itemize}
\setlength{\itemsep}{1pt}
\item \textbf{Poids équilibré} = 1 / valeur moyenne du terme sur les dessins de l'étape 1 (\emph{Adaptive Weight} de DeepGD) : avec ce poids, chaque terme pèserait autant que le stress.
\item \textbf{Coefficient = 3\,\% du poids équilibré}, donc chaque métrique écran pèse environ 3\,\% du stress. Les écarts entre coefficients (0,013 à 10,6) viennent seulement des unités : un angle vaut moins de 1, un croisement se compte en pixels.
\item \textbf{Pourquoi 3\,\%} : le plus fort dosage qui coûte moins de 5\,\% de stress sur la validation (essayés : 3, 5, 10, 30, 100\,\%).
\item \textbf{Remplissage de l'écran retiré} : il étirait les dessins, et le stress en pixels remplit déjà le cadre.
\end{itemize}
\end{minipage}

\begin{frame}{Suite : une taille d'image variable}
\small Aujourd'hui, la taille de l'image est figée dans les poids : la tête a exactement $W \times H$ sorties.

\vspace{0.3cm}
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{block}{Grille normalisée}
\small Le modèle dessine dans $[0,1]^2$, puis on convertit en pixels à la sortie :\\
$x_{\text{px}} = m + x \cdot (W - 2m)$.\\
La taille de l'image devient une entrée, plus un hyperparamètre.
\end{block}
\end{column}
\begin{column}{0.48\textwidth}
\begin{block}{Dire au modèle pour quelle image il dessine}
\small Lui donner le rapport $W/H$ et la taille des nœuds rapportée à l'image ($r/W$), et tirer des tailles d'image variées pendant l'entraînement.
\end{block}
\end{column}
\end{columns}

\vspace{0.3cm}
\scriptsize\textcolor{gris}{Option plus ambitieuse : un réseau qui donne un score pour n'importe quel point $(x, y)$, évalué sur la grille voulue. Plus souple, mais il faut un calcul par nœud et par pixel.}
\end{frame}

%---------------------------------------------------------------------
\begin{frame}{Suite : rendre le modèle attentif aux arêtes}
\small Les arêtes ne sont pas prédites, ce sont des segments entre les nœuds. Pourtant, la plupart des défauts à l'écran viennent d'elles.

\vspace{0.3cm}
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{block}{A. Tête auxiliaire}
\small Une tête en plus prédit les défauts de chaque arête (croisements, nœud caché, longueur). Entraînée par une perte auxiliaire, elle force le modèle à encoder l'information sur les arêtes.
\end{block}
\end{column}
\begin{column}{0.48\textwidth}
\begin{block}{B. Raffinement itératif}
\small Une première passe propose des positions. On calcule exactement les défauts de chaque arête, puis une seconde passe les reçoit en entrée et corrige.
\end{block}
\end{column}
\end{columns}
\end{frame}

%---------------------------------------------------------------------
\begin{frame}{Bonus : affiner le modèle selon nos préférences}
\small On annote des paires de dessins (\og je préfère A à B\fg{}), on apprend un critique sur l'image (un CNN), puis on affine le modèle pour lui plaire, en le gardant proche du modèle de départ pour qu'il ne triche pas.

  \node[carte,text width=4.15cm,draw=vert,line width=0.8pt,right=of a] (b) {\textcolor{vert}{\scriptsize OPTION 2}\\\textbf{DPO}\\[2pt]\scriptsize directement à partir des paires, sans critique ni boucle de RL};

\end{frame}
